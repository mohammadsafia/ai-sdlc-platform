# QA Stage (5) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Structured test plans per BRD, case-level AI QA results, human QA sign-off with a merge warning, and a per-requirement QA rollup.

**Architecture:** `docs/qa/<brd>.testplan.json` handled by `src/main/qa/` and `src/shared/qa/` modules that mirror the requirements set. A structured-output generator produces cases per milestone. The worker injects the task's approved cases into the QA agents' prompts; the QA loop stores `case_results` from the sign-off in `task_metadata.json`. A sign-off panel in task review writes human results through one IPC call. A Test plan tab in the BRD editor, a merge warning dialog, and a rollup line complete the renderer side.

**Tech Stack:** Electron main + preload + React renderer, TypeScript strict, Zustand, react-i18next, Vitest, Biome, Vercel AI SDK `generateText` + `Output.object`.

**Spec:** `docs/superpowers/specs/2026-09-22-qa-stage-design.md`

## Global Constraints

- Paths relative to `apps/desktop/`; run commands there. Work on `develop`, commit per task.
- Vercel AI SDK only, via `createSimpleClient` + `generateText`, as `requirements-generator.ts` does.
- All UI text through `react-i18next`; keys in `en` and `fr`.
- Plan file `docs/qa/<brd-slug>.testplan.json`; case ids `^TC\d+$`; statuses `draft | approved`; result statuses `passed | failed | skipped`.
- Sign-off key in `implementation_plan.json.qa_signoff` is `case_results` (never `test_results`, which the schema aliases to `tests_passed`).
- Injection agent types: `qa_reviewer`, `qa_fixer`; cap 40 KB.
- Error strings: `Generate a requirements set before a test plan`, `Milestone <id> not found`, `A run is already in progress for this BRD`, `No running generation with id <runId>`, `Unknown test case: <id>`, `Every case needs a status`.
- Human sign-off never blocks merge or done; it only warns.

```ts
// apps/desktop/src/shared/qa/testplan.ts
import { z } from 'zod';

import type { TaskMetadata } from '../types/task';
import { CASE_RESULT_STATUSES, TEST_CASE_KINDS, type CaseResult, type TestCase, type TestPlan } from '../types/qa';
import type { RequirementsSet } from '../types/requirements';

const CASE_ID = /^TC\d+$/;

export const TestCaseSchema = z.object({
  id: z.string().regex(CASE_ID),
  requirementId: z.string().regex(/^R\d+$/),
  title: z.string(),
  preconditions: z.string().optional(),
  steps: z.array(z.string()),
  expected: z.string(),
  kind: z.enum(TEST_CASE_KINDS as [string, ...string[]]),
  included: z.boolean(),
});

export const TestPlanSchema = z.object({
  version: z.literal(1),
  brdSlug: z.string(),
  status: z.enum(['draft', 'approved']),
  generatedAt: z.string(),
  approvedAt: z.string().optional(),
  cases: z.array(TestCaseSchema),
});

/** Model-facing shape: nullable rather than optional (OpenAI strict mode). */
export const GeneratedCaseSchema = z.object({
  id: z.string().nullable(),
  requirementId: z.string(),
  title: z.string(),
  preconditions: z.string().nullable(),
  steps: z.array(z.string()),
  expected: z.string(),
  kind: z.enum(TEST_CASE_KINDS as [string, ...string[]]),
});
export const GeneratedCasesSchema = z.object({ cases: z.array(GeneratedCaseSchema) });
export type GeneratedCase = z.infer<typeof GeneratedCaseSchema>;

export function nextCaseId(existing: Array<{ id: string }>): string {
  const max = existing.reduce((m, c) => Math.max(m, Number(c.id.slice(2)) || 0), 0);
  return `TC${max + 1}`;
}

/** Give generated cases ids (keeping echoed ids that exist), mark them included, and drop cases outside the allowed requirements. */
export function assignCaseIds(generated: GeneratedCase[], existing: TestCase[], allowedRequirementIds: string[]): { cases: TestCase[]; warnings: string[] } {
  const known = new Set(existing.map((c) => c.id));
  const used: Array<{ id: string }> = [...existing];
  const cases: TestCase[] = [];
  const warnings: string[] = [];
  for (const g of generated) {
    if (!allowedRequirementIds.includes(g.requirementId)) {
      warnings.push(`Dropped a case for ${g.requirementId}, which is not in this milestone`);
      continue;
    }
    const id = g.id && known.has(g.id) && !cases.some((c) => c.id === g.id) ? g.id : nextCaseId(used);
    used.push({ id });
    cases.push({
      id,
      requirementId: g.requirementId,
      title: g.title,
      ...(g.preconditions ? { preconditions: g.preconditions } : {}),
      steps: g.steps,
      expected: g.expected,
      kind: g.kind as TestCase['kind'],
      included: true,
    });
  }
  return { cases, warnings };
}

function sortCases(cases: TestCase[], requirementOrder: string[]): TestCase[] {
  const rank = (id: string) => {
    const i = requirementOrder.indexOf(id);
    return i === -1 ? Number.MAX_SAFE_INTEGER : i;
  };
  return [...cases].sort((a, b) => rank(a.requirementId) - rank(b.requirementId) || Number(a.id.slice(2)) - Number(b.id.slice(2)));
}

/** Replace the cases of the given requirements with `generated`, keep everything else, and return a draft plan. */
export function mergeMilestoneCases(plan: TestPlan, milestoneRequirementIds: string[], generated: TestCase[], requirementOrder: string[]): TestPlan {
  const kept = plan.cases.filter((c) => !milestoneRequirementIds.includes(c.requirementId));
  const { approvedAt: _a, ...rest } = plan;
  return { ...rest, status: 'draft', generatedAt: new Date().toISOString(), cases: sortCases([...kept, ...generated], requirementOrder) };
}

export function validateTestPlan(plan: TestPlan, set: RequirementsSet | null): string[] {
  const warnings: string[] = [];
  const known = new Set(set?.requirements.map((r) => r.id) ?? []);
  for (const c of plan.cases) {
    if (c.steps.filter((s) => s.trim()).length === 0) warnings.push(`${c.id} has no steps`);
    if (!c.expected.trim()) warnings.push(`${c.id} has no expected result`);
    if (set && !known.has(c.requirementId)) warnings.push(`${c.id} points to ${c.requirementId}, which is not in the requirements set`);
  }
  for (const r of set?.requirements ?? []) {
    if (r.included && r.acceptanceCriteria.length > 0 && !plan.cases.some((c) => c.included && c.requirementId === r.id)) {
      warnings.push(`${r.id} has acceptance criteria but no test case`);
    }
  }
  return warnings;
}

/** Included cases of an approved plan for the given requirements, in plan order. */
export function casesForTask(plan: TestPlan | null, requirementIds: string[]): TestCase[] {
  if (!plan || plan.status !== 'approved') return [];
  return plan.cases.filter((c) => c.included && requirementIds.includes(c.requirementId));
}

type TaskLike = { metadata?: TaskMetadata };

/** AI results overlaid by human results for one task. */
export function mergedResults(task: TaskLike): Map<string, CaseResult> {
  const out = new Map<string, CaseResult>();
  for (const r of task.metadata?.qa?.ai?.results ?? []) out.set(r.caseId, r);
  for (const r of task.metadata?.qa?.human?.results ?? []) out.set(r.caseId, r);
  return out;
}

/** Per requirement: included cases, and how many passed or failed by their latest result across the covering tasks. */
export function qaRollup(plan: TestPlan | null, tasks: TaskLike[]): Record<string, { total: number; passed: number; failed: number }> {
  const out: Record<string, { total: number; passed: number; failed: number }> = {};
  if (!plan) return out;
  const latest = new Map<string, CaseResult>();
  const humanSeen = new Set<string>();
  const aiAt = new Map<string, string>();
  for (const task of tasks) {
    for (const r of task.metadata?.qa?.human?.results ?? []) {
      if (!humanSeen.has(r.caseId)) {
        humanSeen.add(r.caseId);
        latest.set(r.caseId, r);
      }
    }
    const ai = task.metadata?.qa?.ai;
    for (const r of ai?.results ?? []) {
      if (humanSeen.has(r.caseId)) continue;
      const prev = aiAt.get(r.caseId);
      if (!prev || (ai?.at ?? '') > prev) {
        aiAt.set(r.caseId, ai?.at ?? '');
        latest.set(r.caseId, r);
      }
    }
  }
  for (const c of plan.cases) {
    if (!c.included) continue;
    const entry = (out[c.requirementId] ??= { total: 0, passed: 0, failed: 0 });
    entry.total++;
    const r = latest.get(c.id);
    if (r?.status === 'passed') entry.passed++;
    if (r?.status === 'failed') entry.failed++;
  }
  return out;
}

/** Cases the human has not signed yet. */
export function unsignedCases(cases: TestCase[], task: TaskLike): TestCase[] {
  const human = new Set((task.metadata?.qa?.human?.results ?? []).map((r) => r.caseId));
  return cases.filter((c) => c.included && !human.has(c.id));
}

export function isCaseResultStatus(value: unknown): value is CaseResult['status'] {
  return typeof value === 'string' && (CASE_RESULT_STATUSES as readonly string[]).includes(value);
}
```

```ts
// apps/desktop/src/main/qa/testplan-files.ts
import { promises as fs } from 'node:fs';
import path from 'node:path';

import { TestPlanSchema } from '../../shared/qa/testplan';
import type { TestPlan } from '../../shared/types/qa';
import { brdPath } from '../brd/brd-files';

export const QA_DIR = 'docs/qa';

/** `<project>/docs/qa/<slug>.testplan.json`, reusing brdPath's slug validation; containment like brd-files. */
export async function testPlanPath(projectDir: string, slug: string): Promise<string> {
  await brdPath(projectDir, slug); // validates the slug and the project root
  const realProject = await fs.realpath(projectDir);
  const dir = path.join(realProject, QA_DIR);
  try {
    const real = await fs.realpath(dir);
    const boundary = realProject.endsWith(path.sep) ? realProject : realProject + path.sep;
    if (!real.startsWith(boundary)) throw new Error(`${QA_DIR} resolves outside the project`);
  } catch (err) {
    if ((err as NodeJS.ErrnoException).code !== 'ENOENT') throw err;
  }
  return path.join(dir, `${slug}.testplan.json`);
}

export async function readTestPlan(projectDir: string, slug: string): Promise<TestPlan | null> {
  const file = await testPlanPath(projectDir, slug);
  let raw: string;
  try {
    raw = await fs.readFile(file, 'utf-8');
  } catch (err: unknown) {
    if ((err as NodeJS.ErrnoException).code === 'ENOENT') return null;
    throw err;
  }
  let json: unknown;
  try {
    json = JSON.parse(raw);
  } catch {
    throw new Error(`${slug}.testplan.json is invalid: not JSON`);
  }
  const parsed = TestPlanSchema.safeParse(json);
  if (!parsed.success) {
    throw new Error(`${slug}.testplan.json is invalid: ${parsed.error.issues.map((i) => `${i.path.join('.')}: ${i.message}`).join('; ')}`);
  }
  return parsed.data as TestPlan;
}

export async function writeTestPlan(projectDir: string, slug: string, plan: TestPlan): Promise<TestPlan> {
  const file = await testPlanPath(projectDir, slug);
  await fs.mkdir(path.dirname(file), { recursive: true });
  const tmp = `${file}.tmp-${process.pid}-${Date.now()}`;
  await fs.writeFile(tmp, `${JSON.stringify(plan, null, 2)}\n`, 'utf-8');
  await fs.rename(tmp, file);
  return plan;
}
```

```ts
export function buildTestPlanPrompts(config: TestPlanRunConfig, projectContext: string): { system: string; prompt: string } {
  const rules = tryLoadPrompt('testplan_generator') ?? FALLBACK_SYSTEM;
  const system = `${rules.trim()}\n` + (projectContext ? `\n## PROJECT CONTEXT\n\n${projectContext.trim()}\n` : '');

  for (const r of config.requirements) {
    parts.push(`### ${r.id}: ${r.title}`, `Area: ${r.area}`, r.description.trim(), '', 'Acceptance criteria:', ...r.acceptanceCriteria.map((c) => `- ${c}`), '');
  }
  if (config.briefs.length > 0) {
    parts.push('## DESIGN BRIEFS', '');
    for (const b of config.briefs) parts.push(`### ${b.requirementId}`, b.body.trim(), '');
  }
  if (config.existingCases.length > 0) {
    const stripped = config.existingCases.map(({ included: _i, ...c }) => c);
    parts.push('## EXISTING CASES', '', JSON.stringify(stripped, null, 2), '');
  }
  parts.push('## BRD', '', config.brdMarkdown.trim(), '', 'Produce the test cases as JSON now.');
  return { system, prompt: parts.join('\n') };

function withNullDefaults(raw: unknown): unknown {
  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return raw;
  const body = { ...(raw as Record<string, unknown>) };
  if (Array.isArray(body.cases)) {
    body.cases = body.cases.map((c) =>
      c && typeof c === 'object' && !Array.isArray(c)
        ? { id: null, preconditions: null, ...(c as Record<string, unknown>) }
        : c,
    );
  }
  return body;
}

function safeJson(text: string): unknown {
  try {
    return JSON.parse(text);
  } catch {
    return null;
  }
}

export async function runTestPlanGenerator(config: TestPlanRunConfig, onEvent: (e: TestPlanRunEvent) => void): Promise<void> {
  onEvent({ type: 'progress', phase: 'started' });
  try {
    const { system, prompt } = buildTestPlanPrompts(config, loadBrdProjectContext(config.projectDir));
    const client = await createSimpleClient({
      systemPrompt: system,
      modelShorthand: config.modelShorthand ?? 'sonnet',
      thinkingLevel: config.thinkingLevel ?? 'medium',
      maxSteps: 1,
      tools: {},
    });
    const call = async (userPrompt: string) => {
      const result = await generateText({
        model: client.model,
        system,
        prompt: userPrompt,
        abortSignal: config.abortSignal,
        output: Output.object({ schema: GeneratedCasesSchema }),
      });
      // biome-ignore lint/suspicious/noExplicitAny: result.output type varies with the OUTPUT generic
      const anyResult = result as any;
      const direct = anyResult.output != null ? GeneratedCasesSchema.safeParse(withNullDefaults(anyResult.output)) : null;
      if (direct?.success) return { cases: direct.data.cases as GeneratedCase[], text: String(anyResult.text ?? '') };
      onEvent({ type: 'progress', phase: 'parsing' });
      const text = String(anyResult.text ?? '');
      const raw = parseLLMJson(text, z.unknown());
      const parsed = raw == null ? null : GeneratedCasesSchema.safeParse(withNullDefaults(raw));
      return { cases: parsed?.success ? (parsed.data.cases as GeneratedCase[]) : null, text };
    };

    const first = await call(prompt);
    if (first.cases) {
      onEvent({ type: 'done', cases: first.cases });
      return;
    }
    onEvent({ type: 'progress', phase: 'repairing' });
    const validation = GeneratedCasesSchema.safeParse(withNullDefaults(safeJson(first.text)));
    const errors = validation.success ? ['Output was not valid JSON'] : formatZodErrors(validation.error);
    const retry = await call(`${prompt}\n\n${buildValidationRetryPrompt('test plan', errors)}`);
    if (retry.cases) {
      onEvent({ type: 'done', cases: retry.cases });
      return;
    }
    onEvent({ type: 'error', error: 'The model did not return a valid test plan after one retry' });
  } catch (err: unknown) {
    onEvent({ type: 'error', error: err instanceof Error ? err.message : String(err) });
  }
}
```

```ts
/**
 * Merge a QA patch into task_metadata.json's `qa` block (ai or human results).
 * Creates the file when missing. Returns false on failure (logged, never throws).
 */
export function updateTaskMetadataQa(metadataPath: string, patch: Partial<TaskQa>): boolean {
  try {
    let metadata: Record<string, unknown> = {};
    try {
      metadata = safeParseJson<Record<string, unknown>>(readFileSync(metadataPath, 'utf-8')) || {};
    } catch (err) {
      if (!isFileNotFoundError(err)) throw err;
    }
    const current = (metadata.qa && typeof metadata.qa === 'object' ? metadata.qa : {}) as TaskQa;
    metadata.qa = { ...current, ...patch };
    mkdirSync(path.dirname(metadataPath), { recursive: true });
    writeFileAtomicSync(metadataPath, JSON.stringify(metadata, null, 2));
    return true;
  } catch (err) {
    console.warn(`[plan-file-utils] Could not update QA metadata at ${metadataPath}:`, err);
    return false;
  }
}
```

```ts
/**
 * Approved test cases for the requirements a task covers, rendered for the QA agents.
 */
import { existsSync, readFileSync } from 'node:fs';
import { join } from 'node:path';

import { casesForTask } from '../../../shared/qa/testplan';
import type { TestCase } from '../../../shared/types/qa';
import { readTestPlan } from '../../qa/testplan-files';

export const QA_AGENT_TYPES: ReadonlySet<string> = new Set(['qa_reviewer', 'qa_fixer']);
export const QA_SECTION_CAP = 40 * 1024;

const HEADER =
  '# Test cases to verify\n\n' +
  'The QA team approved these cases for the requirements this task covers. Verify each one and record the outcome in implementation_plan.json under qa_signoff.case_results as an array of { "case": "TC3", "status": "passed" | "failed" | "skipped", "note": "..." }. A failed case is an issue to report; a skipped case needs a note saying why.\n\n';

export function renderQaSection(cases: TestCase[], cap: number = QA_SECTION_CAP): string {
  if (cases.length === 0) return '';
  let out = HEADER;
  const omitted: string[] = [];
  for (const c of cases) {
    const lines = [`## ${c.id} (${c.requirementId}): ${c.title} [${c.kind}]`];
    if (c.preconditions) lines.push(`Preconditions: ${c.preconditions}`);
    lines.push('Steps:', ...c.steps.map((s, i) => `${i + 1}. ${s}`), `Expected: ${c.expected}`, '');
    const block = `${lines.join('\n')}\n`;
    if (omitted.length > 0 || out.length + block.length > cap) {
      omitted.push(c.id);
      continue;
    }
    out += block;
  }
  if (omitted.length > 0) out += `(${omitted.length} cases omitted: ${omitted.join(', ')})\n\n`;
  return out;
}

function readLinks(specDir: string): { brdSlug: string; requirementIds: string[] } | null {
  const file = join(specDir, 'task_metadata.json');
  if (!existsSync(file)) return null;
  try {
    const meta = JSON.parse(readFileSync(file, 'utf-8')) as { brdSlug?: unknown; requirementIds?: unknown };
    if (typeof meta.brdSlug !== 'string' || !Array.isArray(meta.requirementIds) || meta.requirementIds.length === 0) return null;
    return { brdSlug: meta.brdSlug, requirementIds: meta.requirementIds.filter((id): id is string => typeof id === 'string') };
  } catch {
    return null;
  }
}

/** Section text, or '' for other agents, unlinked tasks, draft plans, or read errors. Never throws. */
export async function buildQaSectionForAgent(projectDir: string, specDir: string, agentType: string): Promise<string> {
  if (!QA_AGENT_TYPES.has(agentType)) return '';
  const links = readLinks(specDir);
  if (!links) return '';
  try {
    return renderQaSection(casesForTask(await readTestPlan(projectDir, links.brdSlug), links.requirementIds));
  } catch {
    return '';
  }
}
```

```ts
// apps/desktop/src/renderer/stores/testplan-store.ts
import { create } from 'zustand';

import { nextCaseId, validateTestPlan } from '../../shared/qa/testplan';
import type { TestCase, TestPlan, TestPlanRunPhase } from '../../shared/types/qa';
import { useRequirementsStore } from './requirements-store';

interface Proposal { plan: TestPlan; warnings: string[] }
interface RunState { status: 'idle' | 'running' | 'proposal'; runId?: string; phase?: TestPlanRunPhase; proposal?: Proposal; error?: string }

interface TestPlanState {
  slug: string | null;
  plan: TestPlan | null;
  savedPlan: TestPlan | null;
  warnings: string[];
  run: RunState;
  isLoading: boolean;
  isSaving: boolean;
  error: string | null;

  isDirty: () => boolean;
  canApprove: () => boolean;
  reset: () => void;
  load: (projectId: string, slug: string) => Promise<void>;
  editCase: (id: string, patch: Partial<TestCase>) => void;
  toggleInclude: (id: string) => void;
  addCase: (requirementId: string) => void;
  removeCase: (id: string) => void;
  save: (projectId: string) => Promise<void>;
  approve: (projectId: string) => Promise<void>;
  generate: (projectId: string, milestoneId: string) => Promise<void>;
  cancel: () => Promise<void>;
  accept: () => void;
  discard: () => void;
}

const idle: RunState = { status: 'idle' };
const initial = {
  slug: null as string | null,
  plan: null as TestPlan | null,
  savedPlan: null as TestPlan | null,
  warnings: [] as string[],
  run: idle,
  isLoading: false,
  isSaving: false,
  error: null as string | null,
};

const same = (a: TestPlan | null, b: TestPlan | null) => JSON.stringify(a) === JSON.stringify(b);
const warningsFor = (plan: TestPlan | null) => (plan ? validateTestPlan(plan, useRequirementsStore.getState().set) : []);

export const useTestPlanStore = create<TestPlanState>((set, get) => {
  const update = (mutate: (p: TestPlan) => TestPlan) => {
    const current = get().plan;
    if (!current) return;
    const next = mutate(current);
    set({ plan: next, warnings: warningsFor(next) });
  };

  return {
    ...initial,

    isDirty: () => !same(get().plan, get().savedPlan),
    canApprove: () => {
      const { plan, warnings } = get();
      return !!plan && plan.status !== 'approved' && plan.cases.length > 0 && !get().isDirty() && warnings.length === 0;
    },

    reset: () => set({ ...initial, run: { ...idle } }),

    load: async (projectId, slug) => {
      set({ isLoading: true, error: null, slug, run: { ...idle } });
      const result = await window.electronAPI.testPlanRead(projectId, slug);
      if (!result.success) {
        set({ error: result.error ?? 'Unknown error', isLoading: false });
        return;
      }
      const plan = result.data ?? null;
      set({ plan, savedPlan: plan, warnings: warningsFor(plan), isLoading: false });
    },

    editCase: (id, patch) => update((p) => ({ ...p, cases: p.cases.map((c) => (c.id === id ? { ...c, ...patch } : c)) })),
    toggleInclude: (id) => update((p) => ({ ...p, cases: p.cases.map((c) => (c.id === id ? { ...c, included: !c.included } : c)) })),
    addCase: (requirementId) => {
      const slug = get().slug;
      if (!slug) return;
      const base: TestPlan = get().plan ?? { version: 1, brdSlug: slug, status: 'draft', generatedAt: new Date().toISOString(), cases: [] };
      const next: TestPlan = {
        ...base,
        cases: [...base.cases, { id: nextCaseId(base.cases), requirementId, title: '', steps: [], expected: '', kind: 'manual', included: true }],
      };
      set({ plan: next, warnings: warningsFor(next) });
    },
    removeCase: (id) => update((p) => ({ ...p, cases: p.cases.filter((c) => c.id !== id) })),

    save: async (projectId) => {
      const { slug, plan } = get();
      if (!slug || !plan) return;
      set({ isSaving: true, error: null });
      const result = await window.electronAPI.testPlanWrite(projectId, slug, plan);
      if (!result.success || !result.data) {
        set({ error: result.error ?? 'Unknown error', isSaving: false });
        return;
      }
      set({ plan: result.data, savedPlan: result.data, warnings: warningsFor(result.data), isSaving: false });
    },

    approve: async (projectId) => {
      const { slug, plan } = get();
      if (!slug || !plan || !get().canApprove()) return;
      set({ isSaving: true, error: null });
      const result = await window.electronAPI.testPlanApprove(projectId, slug, plan);
      if (!result.success || !result.data) {
        set({ error: result.error ?? 'Unknown error', isSaving: false });
        return;
      }
      set({ plan: result.data, savedPlan: result.data, isSaving: false });
    },

    generate: async (projectId, milestoneId) => {
      const { slug } = get();
      if (!slug) return;
      set({ run: { status: 'running', phase: 'started' } });
      const result = await window.electronAPI.testPlanGenerate(projectId, { slug, milestoneId });
      if (!result.success || !result.data) {
        set({ run: { status: 'idle', error: result.error ?? 'Unknown error' } });
        return;
      }
      set({ run: { status: 'running', runId: result.data.runId, phase: 'started' } });
    },

    cancel: async () => {
      const { runId } = get().run;
      if (runId) await window.electronAPI.testPlanCancel(runId);
    },

    accept: () => {
      const { run } = get();
      if (run.status !== 'proposal' || !run.proposal) return;
      set({ plan: run.proposal.plan, warnings: warningsFor(run.proposal.plan), run: { ...idle } });
    },
    discard: () => set({ run: { ...idle } }),
  };
});

/** Subscribe to generation events. Returns an unsubscribe function. */
export function setupTestPlanListeners(): () => void {
  const store = useTestPlanStore;
  const isCurrent = (runId: string) => store.getState().run.runId === runId;
  const offProgress = window.electronAPI.onTestPlanProgress(({ runId, phase }) => {
    if (!isCurrent(runId)) return;
    store.setState((s) => ({ run: { ...s.run, phase } }));
  });
  const offDone = window.electronAPI.onTestPlanDone(({ runId, plan, warnings }) => {
    if (!isCurrent(runId)) return;
    store.setState({ run: { status: 'proposal', runId, proposal: { plan, warnings } } });
  });
  const offError = window.electronAPI.onTestPlanError(({ runId, error }) => {
    if (!isCurrent(runId)) return;
    store.setState({ run: { status: 'idle', error } });
  });
  return () => {
    offProgress();
    offDone();
    offError();
  };
}
```

```tsx
import { useTranslation } from 'react-i18next';

import type { TestPlan } from '../../../../shared/types/qa';
import { useTestPlanStore } from '../../../stores/testplan-store';
import { Button } from '../../ui/button';

/** Counts of the proposal against the current plan: new ids, replaced ids, untouched ids. */
export function proposalCounts(current: TestPlan | null, proposed: TestPlan): { added: number; replaced: number; kept: number } {
  const before = new Map((current?.cases ?? []).map((c) => [c.id, JSON.stringify(c)]));
  let added = 0;
  let replaced = 0;
  let kept = 0;
  for (const c of proposed.cases) {
    const prev = before.get(c.id);
    if (prev === undefined) added++;
    else if (prev !== JSON.stringify(c)) replaced++;
    else kept++;
  }
  return { added, replaced, kept };
}

export function TestPlanProposal() {
  const { t } = useTranslation('qa');
  const { plan, run, accept, discard } = useTestPlanStore();
  if (run.status !== 'proposal' || !run.proposal) return null;
  const counts = proposalCounts(plan, run.proposal.plan);
  return (
    <div className="space-y-2 rounded-md border border-border bg-muted/30 p-3 text-sm">
      <div className="font-medium">{t('proposal.title')}</div>
      <div className="text-xs text-muted-foreground">{t('proposal.summary', { added: counts.added, replaced: counts.replaced, kept: counts.kept })}</div>
      {run.proposal.warnings.length > 0 && (
        <ul className="list-disc pl-5 text-xs text-amber-600">
          {run.proposal.warnings.map((w) => (
            <li key={w}>{w}</li>
          ))}
        </ul>
      )}
      <div className="flex gap-2">
        <Button size="sm" onClick={accept}>{t('proposal.accept')}</Button>
        <Button size="sm" variant="outline" onClick={discard}>{t('proposal.discard')}</Button>
      </div>
    </div>
  );
}
```
